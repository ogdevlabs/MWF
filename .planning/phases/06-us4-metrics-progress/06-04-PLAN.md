This is a ConsumerStatefulWidget (NOT ConsumerWidget) because:

Implementation:
```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:mwf_mobile/features/metrics/data/metric_providers.dart';

/// Bottom sheet for logging a single metric entry.
/// Per D-03: logs ONE metric type at a time; "Log another" resets form inline.
/// Per D-04: date defaults to today, editable via date picker.
class MetricLogBottomSheet extends ConsumerStatefulWidget {
  const MetricLogBottomSheet({super.key});

  @override
  ConsumerState<MetricLogBottomSheet> createState() => _MetricLogBottomSheetState();
}

class _MetricLogBottomSheetState extends ConsumerState<MetricLogBottomSheet> {
  final _formKey = GlobalKey<FormState>();
  final _valueController = TextEditingController();

  // Metric type selector state
  String _selectedType = 'weight';
  String? _selectedSubtype;
  DateTime _selectedDate = DateTime.now();
  bool _isSubmitting = false;
  bool _showSuccess = false;

  // Metric type options per D-10, D-11, D-12
  static const _metricTypes = ['weight', 'measurement', 'flexibility'];
  static const _measurementSubtypes = ['waist', 'hip', 'chest', 'thigh', 'arm'];
  static const _flexibilitySubtypes = ['forward_bend', 'shoulder', 'hip_flexor'];

  // Unit per type per D-10 (kg/lbs for weight), D-11/D-12 (cm, degrees)
  String get _unit {
    switch (_selectedType) {
      case 'weight': return 'kg';
      case 'measurement': return 'cm';
      case 'flexibility': return 'degrees';
      default: return 'kg';
    }
  }

  List<String>? get _subtypes {
    switch (_selectedType) {
      case 'measurement': return _measurementSubtypes;
      case 'flexibility': return _flexibilitySubtypes;
      default: return null;
    }
  }

  @override
  void dispose() {
    _valueController.dispose();
    super.dispose();
  }

  Future<void> _submit() async {
    if (!_formKey.currentState!.validate()) return;
    setState(() => _isSubmitting = true);

    final repo = ref.read(metricRepositoryProvider);
    await repo.logMetric(
      metricType: _selectedType,
      metricSubtype: _selectedSubtype,
      value: double.parse(_valueController.text),
      unit: _unit,
      loggedAt: _selectedDate,
    );

    setState(() {
      _isSubmitting = false;
      _showSuccess = true;
    });
  }

  void _logAnother() {
    _formKey.currentState!.reset();
    _valueController.clear();
    setState(() {
      _selectedSubtype = null;
      _selectedDate = DateTime.now();
      _showSuccess = false;
    });
  }

  Future<void> _pickDate() async {
    final picked = await showDatePicker(
      context: context,
      initialDate: _selectedDate,
      firstDate: DateTime.now().subtract(const Duration(days: 365)),
      lastDate: DateTime.now(),
    );
    if (picked != null) {
      setState(() => _selectedDate = picked);
    }
  }

  @override
  Widget build(BuildContext context) { ... }
  // Build method renders:
  // 1. DragHandle at top
  // 2. Title: "Log Metric"
  // 3. If _showSuccess: success message + "Log another" TextButton + "Done" FilledButton
  // 4. If !_showSuccess: Form with:
  //    a. DropdownButtonFormField for metric type (_metricTypes)
  //    b. If _subtypes != null: DropdownButtonFormField for subtype
  //    c. TextFormField for value (keyboardType: TextInputType.numberWithOptions(decimal: true))
  //       validator: must be non-empty, must parse to double > 0
  //    d. Date row: Text showing formatted date + TextButton "Change" triggering _pickDate()
  //    e. FilledButton "Log" calling _submit()
  //    f. TextButton "Cancel" calling Navigator.pop(context)
  // Wrap in Padding(padding: EdgeInsets.only(bottom: MediaQuery.of(context).viewInsets.bottom))
  // for keyboard avoidance
}
```
